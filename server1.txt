1) Password Strength Testing using Python

import re

def check_password_strength(password):
    if len(password) < 8:
        return "Weak: Password must be at least 8 characters long."
    if not any(char.isdigit() for char in password):
        return "Weak: Password must include at least one number."
    if not any(char.isupper() for char in password):
        return "Weak: Password must include at least one uppercase letter."
    if not any(char.islower() for char in password):
        return "Weak: Password must include at least one lowercase letter."
    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        return "Medium: Add special characters to make your password stronger."
    return "Strong: Your password is secure!"

def password_checker():
    print("Welcome to the Password Strength Checker!")
    while True:
        password = input("\nEnter your password (or type 'exit' to quit): ")
        if password.lower() == "exit":
            print("Thank you for using the Password Strength Checker! Goodbye!")
            break
        result = check_password_strength(password)
        print(result)

if __name__ == "__main__":
    password_checker()

Write a Python program that downloads a list of malicious IP addresses from a trusted online
source and automatically creates firewall rules to block those IPs from accessing the system.


import requests
import csv
import subprocess

def update_firewall_blocklist():
    url = "https://feodotracker.abuse.ch/downloads/ipblocklist.csv"
    rule_name = "BadIP"

    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        
        lines = response.text.splitlines()
        data = filter(lambda x: not x.startswith("#"), lines)
        mycsv = csv.reader(data)
        
        ip_list = []
        for row in mycsv:
            if row and row[1] != "dst_ip":
                ip_list.append(row[1])

        if not ip_list:
            print("No IPs found to block.")
            return
        subprocess.run(["powershell", "-Command", f'netsh advfirewall firewall delete rule name="{rule_name}"'], 
                       capture_output=True)

        remote_ips = ",".join(ip_list)
        rule_cmd = f'netsh advfirewall firewall add rule name="{rule_name}" Dir=Out Action=Block RemoteIP={remote_ips}'
        
        print(f"Adding {len(ip_list)} IPs to firewall rule '{rule_name}'...")
        result = subprocess.run(["powershell", "-Command", rule_cmd], capture_output=True, text=True)

        if result.returncode == 0:
            print("Successfully updated firewall rules.")
        else:
            print(f"Error adding rule: {result.stderr}")

    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    update_firewall_blocklist()

Execution Steps:

    Search for rule name: BadIP
    You will see many blocked IP addresses listed
